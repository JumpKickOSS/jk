--- jk.toml
group = "com.acme"
name = "demo"
version = "…"
# Helidon 27 is compiled for Java 27; the first jk build installs that JDK when the machine has none.
java = 27

[application]
main = "com.acme.Main"

# The Helidon BOM (io.helidon:helidon-dependencies), so the dependencies below carry no version;
# Helidon's annotation processors; the test config profile; and Helidon's deployable,
# target/demo.jar with target/libs/.
[helidon]
version = "…"

[dependencies]
helidon-webserver = { group = "io.helidon.webserver", name = "helidon-webserver" }
helidon-config-yaml = { group = "io.helidon.config", name = "helidon-config-yaml" }
helidon-http-media-jsonp = { group = "io.helidon.http.media", name = "helidon-http-media-jsonp" }
# Reads the Pokemon record a POST or PUT carries.
helidon-http-media-jsonb = { group = "io.helidon.http.media", name = "helidon-http-media-jsonb" }
jakarta-json-api = { group = "jakarta.json", name = "jakarta.json-api" }
helidon-dbclient = { group = "io.helidon.dbclient", name = "helidon-dbclient" }
# A database row as a JSON-P object: row.as(JsonObject.class).
helidon-dbclient-jsonp = { group = "io.helidon.dbclient", name = "helidon-dbclient-jsonp" }
helidon-dbclient-mongodb = { group = "io.helidon.dbclient", name = "helidon-dbclient-mongodb" }
[runtime-dependencies]
helidon-logging-jul = { group = "io.helidon.logging", name = "helidon-logging-jul" }
# HikariCP and the MongoDB driver log through SLF4J; this sends it to java.util.logging.
slf4j-jdk14 = { group = "org.slf4j", name = "slf4j-jdk14" }
[test-dependencies]
junit-jupiter = { group = "org.junit.jupiter", name = "junit-jupiter" }
helidon-webserver-testing-junit5 = { group = "io.helidon.webserver.testing.junit5", name = "helidon-webserver-testing-junit5" }
hamcrest-all = { group = "org.hamcrest", name = "hamcrest-all" }
# The tests run on an in-memory H2 (application-test.yaml).
h2 = { group = "com.h2database", name = "h2" }
helidon-dbclient-jdbc = { group = "io.helidon.dbclient", name = "helidon-dbclient-jdbc" }
helidon-dbclient-hikari = { group = "io.helidon.dbclient", name = "helidon-dbclient-hikari" }

# jk build also writes target/demo-jri/: a runtime linked from the project JDK with the modules
# jdeps finds, the application under app/, bin/start, and an AOT cache trained by starting it.
[jlink]
# HikariCP and the database drivers load through ServiceLoader, where jdeps does not look, and use JNDI.
modules = ["java.naming"]
# The build's starts of the image (the AOT training and the start test) start the application,
# which needs its database up; jk build would fail without it.
aot-cache  = false
test-start = false

# Null checking: every package is @NullMarked (package-info.java), and NullAway checks it at
# compile time. The table brings JSpecify, Error Prone and NullAway at versions jk tests together.
[nullness]

# Test tiers: `jk test` is the fast tier; each profile owns one cost tag, so every tagged
# test runs in exactly one tier (`jk test -p integration`).
[test]
exclude-tags = ["integration", "network", "slow", "bench"]

[profiles.integration]
include-tags = ["integration"]
exclude-tags = []

[profiles.network]
include-tags = ["network"]
exclude-tags = ["integration"]

[profiles.slow]
include-tags = ["slow"]
exclude-tags = ["integration", "network"]

[profiles.bench]
include-tags = ["bench"]
exclude-tags = ["integration", "network", "slow"]
