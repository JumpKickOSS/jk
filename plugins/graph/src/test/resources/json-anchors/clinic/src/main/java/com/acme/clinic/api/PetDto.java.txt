package com.acme.clinic.api;

import com.fasterxml.jackson.annotation.JsonIgnore;

public class PetDto {

    private int id;
    private String name;
    private String type;
    private String chip;
    private String secret;

    public int getId() {
        return id;
    }

    public String getName() {
        return name;
    }

    public String getType() {
        return type;
    }

    @JsonIgnore
    public String getSecret() {
        return secret;
    }
}
