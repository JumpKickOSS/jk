export interface Owner {
  id: number;
  name: string;
  birth_date: string;
  pets: Pet[];
}

export interface Pet {
  id: number;
  name: string;
  type: string;
}

export async function fetchOwner(id: number): Promise<Owner> {
  const res = await fetch(`/api/owners/${id}`);
  const owner = await res.json();
  console.log(owner.birth_date);
  return owner;
}

export async function createOwner(first: string, last: string) {
  const res = await fetch('/api/owners', {
    method: 'POST',
    body: JSON.stringify({ first_name: first, last_name: last }),
  });
  return res.json();
}

export async function vetNames(): Promise<string[]> {
  const res = await fetch('/api/vets');
  const vets = await res.json();
  return vets.map((v: { name: string }) => v.name);
}

export function label(o: { name: string }) {
  return o.name;
}
