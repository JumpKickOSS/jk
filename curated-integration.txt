# The integration classes that run on every pull request.
#
# `jk test --profile integration` runs more than a hundred classes; the branch gate cannot afford all
# of them, so it runs this subset (scripts/curated-integration.sh, a step of the self-host job) and
# the full profile stays nightly. Membership is a duplicate execution policy, not a JUnit tag: every
# class named here carries `@Tag("integration")` and runs in the nightly profile too, so nothing is
# reclassified to make the gate pass.
#
# One entry per line, five `|`-separated fields:
#
#   module | class | surface | outcomes | why it is merge-critical
#
#   module    workspace module directory, as the root jk.toml lists it — `clients/cli` or `server/engine`.
#   class     fully-qualified test class; its source must exist under that module's src/test/java.
#   surface   one of: wire, spawn, workers, workspace, install, lock.
#   outcomes  comma-separated subset of {success, failure}. Every surface needs both across its
#             entries, and a `failure` claim has to be visible in the class — a failure-shaped
#             assertion, or a test method whose name says what is refused.
#   why       one sentence: what merges broken when this class is not run.
#
# Blank lines and `#` comments are ignored. Guard G63 (`curated-integration`) checks every rule
# above, the wall-clock budget in docs/contributors/test-suite-tiers.md, and the CI step that runs
# the lane.
#
# One setting, `integration-floor = N`: the smallest `@Tag("integration")` population the guard
# accepts from its scan of every src/test/java. Under it, either the tier shrank into the curated
# subset or the scan broke, and the guard refuses to judge rather than pass a lane carved out of
# nothing. A deliberate shrink lowers the number here, in the same commit; without the line the
# floor is 90.

integration-floor = 90
